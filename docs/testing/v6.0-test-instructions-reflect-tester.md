### 2. Search Quality Validation

**Test: Memory Tool Priority in Hybrid Search**
```python
# Create memory file with specific content
await store_to_memory(
    ctx,
    "patterns/search-test.md",
    "# Unique Search Term XYZTEST123\n\nThis should rank highest",
    category="patterns"
)

# Search using hybrid search
results = await csr_reflect_on_past(ctx, "XYZTEST123")

# Memory Tool result should appear first (score 0.95)
first_result = results[0]
assert first_result['source'] == 'memory_tool' or first_result['score'] >= 0.95
```

### 3. File Size Protection

**Test: Large File Handling**
```python
# Create file near 1MB limit
content_900kb = "x" * 900_000
await store_to_memory(ctx, "test/near-limit.md", content_900kb)

# Should be searchable
result = await search_memory(ctx, "x")
assert "test/near-limit.md" in result

# Create file over 1MB limit
content_1_1mb = "x" * 1_100_000
await store_to_memory(ctx, "test/over-limit.md", content_1_1mb)

# Should be skipped in search (check logs for warning)
result = await search_memory(ctx, "x")
# File exists but search skips it
```

### 4. Category Organization

**Test: Memory Organization**
```python
# Create files in different categories
await store_to_memory(ctx, "patterns/p1.md", "Pattern 1", category="patterns")
await store_to_memory(ctx, "insights/i1.md", "Insight 1", category="insights")
await store_to_memory(ctx, "quality/q1.md", "Quality 1", category="quality")

# List by category
patterns = await list_memories(ctx, category="patterns")
assert "patterns/p1.md" in patterns
assert "insights/i1.md" not in patterns

# List all
all_mems = await list_memories(ctx, category=None)
assert "patterns/p1.md" in all_mems
assert "insights/i1.md" in all_mems
assert "quality/q1.md" in all_mems
```

### 5. Dual Storage Validation

**Test: also_reflect Parameter**
```python
# Store with dual storage
result = await store_to_memory(
    ctx,
    "patterns/dual-test.md",
    "# Dual Storage Test\n\nStored in both systems",
    also_reflect=True
)

# Should confirm both storages
assert "✅ Created memory" in result
assert "Stored reflection" in result or "reflection" in result.lower()

# Verify in Memory Tool
mem_result = await view_memory(ctx, "patterns/dual-test.md")
assert "Dual Storage Test" in mem_result

# Verify in Qdrant via search
vec_result = await csr_reflect_on_past(ctx, "Dual Storage Test")
assert len(vec_result) > 0
```

### 6. Security Boundary Tests

**Test: Path Validation**
```python
# Valid paths should work
result = await store_to_memory(ctx, "patterns/valid.md", "Valid")
assert "✅" in result

# Invalid paths should be blocked
result = await store_to_memory(ctx, "../../../etc/passwd", "Evil")
assert "Security error" in result
assert "Path traversal" in result

result = await view_memory(ctx, "/etc/passwd")
assert "Security error" in result or "Memory file not found" in result
```

### 7. Error Handling

**Test: Graceful Failures**
```python
# View non-existent file
result = await view_memory(ctx, "nonexistent.md")
assert "Memory file not found" in result
assert "error" in result.lower()

# Search in non-existent category
result = await list_memories(ctx, category="nonexistent")
# Should return empty or "No memory files found"
assert "No memory files found" in result or result == "[]"

# Update non-existent file
result = await handler.str_replace("nonexistent.md", "old", "new")
assert "Memory file not found" in result
```
