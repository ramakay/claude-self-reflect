use std::collections::HashSet;
use serde::{Deserialize, Serialize};

/// Structured episode extracted from a session transcript.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Episode {
    pub schema: String,
    pub session_id: String,
    pub project: String,
    pub timestamp: String,
    pub request: String,
    pub investigated: Vec<String>,
    pub completed: String,
    pub next_steps: Option<String>,
    pub blockers: Option<String>,
    pub outcome: String,  // "success" | "partial" | "failed" | "interrupted"
    pub error_signatures: Vec<String>,
    pub tools_used: Vec<String>,
    pub files_modified: Vec<String>,
    pub message_count: usize,
    pub duration_minutes: u32,
}

/// Extract a structured episode from transcript lines (JSONL).
/// Pure heuristic — no LLM call. Runs in <10ms.
fn extract_episode(lines: &[&str], session_id: &str, project: &str) -> Episode {
    let mut request = String::new();
    let mut investigated = HashSet::new();
    let mut files_modified = HashSet::new();
    let mut tools_used = HashSet::new();
    let mut last_assistant_text = String::new();
    let mut error_signatures = Vec::new();
    let mut message_count: usize = 0;
    let mut first_ts: Option<&str> = None;
    let mut last_ts: Option<&str> = None;

    for line in lines {
        let Ok(val) = serde_json::from_str::<serde_json::Value>(line) else {
            continue;
        };

        message_count += 1;

        // Track timestamps for duration
        if let Some(ts) = val.pointer("/timestamp").and_then(|v| v.as_str()) {
            if first_ts.is_none() {
                first_ts = Some(ts);
            }
            last_ts = Some(ts);
        }

        let msg_type = val.get("type").and_then(|v| v.as_str()).unwrap_or("");

        // Extract first user message as request
        if (msg_type == "user" || msg_type == "human") && request.is_empty() {
            if let Some(text) = val
                .pointer("/message/content")
                .and_then(|c| c.as_array())
                .and_then(|arr| arr.iter().find(|b| b.get("type").and_then(|t| t.as_str()) == Some("text")))
                .and_then(|b| b.get("text"))
                .and_then(|t| t.as_str())
            {
                request = text.chars().take(200).collect();
            }
        }

        // Extract tool usage from assistant messages
        if msg_type == "assistant" {
            if let Some(content) = val.pointer("/message/content").and_then(|c| c.as_array()) {
                for block in content {
                    let block_type = block.get("type").and_then(|t| t.as_str()).unwrap_or("");

                    if block_type == "tool_use" {
                        if let Some(name) = block.get("name").and_then(|n| n.as_str()) {
                            tools_used.insert(name.to_string());
                        }
                        // Track file paths from tool inputs
                        if let Some(fp) = block.pointer("/input/file_path").and_then(|f| f.as_str()) {
                            let tool_name = block.get("name").and_then(|n| n.as_str()).unwrap_or("");
                            if matches!(tool_name, "Read" | "Glob" | "Grep") {
                                investigated.insert(fp.to_string());
                            }
                            if matches!(tool_name, "Edit" | "Write" | "MultiEdit") {
                                files_modified.insert(fp.to_string());
                            }
                        }
                    }

                    // Track last assistant text (for completed/next_steps)
                    if block_type == "text" {
                        if let Some(text) = block.get("text").and_then(|t| t.as_str()) {
                            if !text.trim().is_empty() {
                                last_assistant_text = text.chars().take(500).collect();
                            }
                        }
                    }

                    // Extract error signatures from tool results
                    if block_type == "tool_result" {
                        if let Some(content_text) = block.get("content").and_then(|c| c.as_str()) {
                            for pattern in &["error[", "Error:", "panic", "FAIL", "FAILED", "Exception"] {
                                if content_text.contains(pattern) {
                                    // Extract the error line (first matching line, truncated)
                                    if let Some(err_line) = content_text.lines().find(|l| l.contains(pattern)) {
                                        let sig: String = err_line.chars().take(100).collect();
                                        if !error_signatures.contains(&sig) {
                                            error_signatures.push(sig);
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    // Determine outcome
    let outcome = if message_count < 3 {
        "interrupted"
    } else if !error_signatures.is_empty() {
        // Check if the LAST message also has errors (ongoing failure vs recovered)
        "failed"
    } else if last_assistant_text.to_lowercase().contains("complete")
        || last_assistant_text.to_lowercase().contains("fixed")
        || last_assistant_text.to_lowercase().contains("done")
        || last_assistant_text.to_lowercase().contains("success")
    {
        "success"
    } else {
        "partial"
    };

    // Extract next_steps from last assistant text
    let next_steps = extract_next_steps(&last_assistant_text);

    // Truncate completed to meaningful portion
    let completed: String = last_assistant_text.chars().take(300).collect();

    // Calculate duration
    let duration_minutes = 0u32; // Simplified — timestamps in JSONL are inconsistent

    Episode {
        schema: "csr_episode_v1".into(),
        session_id: session_id.to_string(),
        project: project.to_string(),
        timestamp: chrono::Utc::now().to_rfc3339(),
        request,
        investigated: investigated.into_iter().collect(),
        completed,
        next_steps,
        blockers: None,
        outcome: outcome.to_string(),
        error_signatures,
        tools_used: tools_used.into_iter().collect(),
        files_modified: files_modified.into_iter().collect(),
        message_count,
        duration_minutes,
    }
}

/// Extract next_steps from assistant text using keyword heuristics.
fn extract_next_steps(text: &str) -> Option<String> {
    let lower = text.to_lowercase();
    let markers = ["next step", "next:", "todo:", "remaining:", "then "];
    for marker in &markers {
        if let Some(pos) = lower.find(marker) {
            let start = pos + marker.len();
            let snippet: String = text[start..].chars().take(200).collect();
            let trimmed = snippet.trim().to_string();
            if !trimmed.is_empty() {
                return Some(trimmed);
            }
        }
    }
    None
}
```

/// Generate tags for episode storage and retrieval.
fn episode_tags(episode: &Episode) -> Vec<String> {
    vec![
        "session_episode".to_string(),
        "schema_v1".to_string(),
        format!("project_{}", episode.project),
        format!("conv_{}", episode.session_id),
    ]
}

/// Store episode as a tagged reflection, replacing any previous episode for the same session.
/// Uses delete+insert (not upsert) to prevent search index noise from stale versions.
fn store_episode(engine: &crate::engine::Engine, episode: &Episode) -> Result<()> {
    let tags = episode_tags(episode);
    let conv_tag = format!("conv_{}", episode.session_id);

    // Delete previous episode for this session (GC)
    let existing = engine
        .storage()
        .get_reflections_by_tag(&conv_tag, 10)
        .unwrap_or_default();
    for (id, _, ref existing_tags, _) in &existing {
        if existing_tags.iter().any(|t| t == "session_episode") {
            let _ = engine.storage().delete_reflection(id);
        }
    }

    // Store new episode as JSON reflection
    let content = serde_json::to_string_pretty(episode)?;

    // Embed synchronously — episode content is short, <2ms
    let embedding = engine.embeddings().embed_single(&content)?;
    engine
        .storage()
        .insert_reflection(&uuid::Uuid::new_v4().to_string(), &content, &tags, &embedding)?;

    Ok(())
}
```

pub async fn handle(input: &HookInput, engine: &Engine, cwd: &Path) -> Result<()> {
    // Import growing transcript for ALL sessions (real-time searchability)
    super::import_current_transcript(input, engine, cwd).await;

    // Write rolling session summary — survives Ctrl+C (session-end may not fire)
    if let Err(e) = write_rolling_summary(input, engine, cwd) {
        eprintln!("CSR: rolling summary error (non-fatal): {}", e);
    }

    // Extract and store structured episode
    if let Err(e) = extract_and_store_episode(input, engine, cwd) {
        eprintln!("CSR: episode extraction error (non-fatal): {}", e);
    }

    Ok(())
}

/// Read transcript and extract+store a structured episode.
fn extract_and_store_episode(input: &HookInput, engine: &Engine, cwd: &Path) -> Result<()> {
    let session_id = match &input.session_id {
        Some(id) => id.clone(),
        None => return Ok(()),
    };
    let transcript_path = match &input.transcript_path {
        Some(p) => p.clone(),
        None => return Ok(()),
    };

    let project = resolve_project_from_cwd(&cwd.to_string_lossy());
    let project_name = project.as_deref().unwrap_or("unknown");

    // Read transcript lines (use BufReader for efficiency)
    let content = std::fs::read_to_string(&transcript_path)?;
    let lines: Vec<&str> = content.lines().collect();

    if lines.is_empty() {
        return Ok(());
    }

    let episode = extract_episode(&lines, &session_id, project_name);

    // Only store if we have a meaningful request (not empty sessions)
    if !episode.request.is_empty() {
        store_episode(engine, &episode)?;
    }

    Ok(())
}
